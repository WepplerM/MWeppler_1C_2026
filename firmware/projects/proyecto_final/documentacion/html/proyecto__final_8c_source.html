299 5,

303 TaskBocina,
304 "TaskBocina",
305 4096,
306 NULL,
307 5,
308 NULL);
309
310 while(1)
311 {
312 switch(BleStatus())
313 {
314 case BLE_OFF:
315 LedOff(LED_BT);

317

321

323 LedOn(LED_BT);
